  drawer.classList.remove("translate-x-full");
  overlay.classList.remove("hidden");

  drawer.classList.add("translate-x-full");
  overlay.classList.add("hidden");

overlay.addEventListener("click", closeCart);

/* ---------- Add To Cart ---------- */
function addToCart(id) {
  const product = products.find(p => p.id === id);

  const existing = cart.find(item => item.id === id);

  if (existing) {
    existing.qty++;
  } else {
    cart.push({ ...product, qty: 1 });
  }

  renderCart();
}

/* ---------- Remove ---------- */
function removeItem(id) {
  cart = cart.filter(item => item.id !== id);
  renderCart();
}

/* ---------- Render ---------- */
function renderCart() {
  cartCount.textContent = cart.reduce((sum, item) => sum + item.qty, 0);

  const cartItems = document.getElementById("cartItems");

  if (!cart.length) {
    cartItems.innerHTML =
      '<p class="text-center text-gray-500">Your cart is empty.</p>';
    document.getElementById("cartTotal").textContent = "Rs 0";
    return;
  }

  let total = 0;

  cartItems.innerHTML = cart.map(item => {
    total += item.price * item.qty;

    return `
      <div class="border rounded-lg p-3">
          <div class="font-semibold">${item.name}</div>

          <div class="text-sm text-gray-500 mt-1">
              Qty: ${item.qty}
          </div>

          <div class="text-blue-600 font-bold">
              Rs ${item.price}
          </div>

          <button
              onclick="removeItem(${item.id})"
              class="mt-2 text-sm bg-red-500 hover:bg-red-600 text-white px-3 py-1 rounded">
              Remove
          </button>
      </div>
    `;
  }).join("");

  document.getElementById("cartTotal").textContent = "Rs " + total;
}

<script>
let discountPercent = 0;

/* Apply Promo Code */
function applyPromo() {
  const code = document.getElementById("promoCode").value.trim().toUpperCase();
  const msg = document.getElementById("promoMsg");

  if (code === "DISCOUNT10") {
    discountPercent = 10;
    msg.innerHTML = "✅ Promo Applied! 10% Discount";
    msg.className = "text-sm mt-2 text-green-600";
  } else {
    discountPercent = 0;
    msg.innerHTML = "❌ Invalid Promo Code";
    msg.className = "text-sm mt-2 text-red-600";
  }

  renderCart();
}

/* Replace your existing renderCart() with this version */

function renderCart() {

  cartCount.textContent = cart.reduce((sum, item) => sum + item.qty, 0);

  const cartItems = document.getElementById("cartItems");

  if (!cart.length) {
    cartItems.innerHTML =
      '<p class="text-center text-gray-500">Your cart is empty.</p>';

    document.getElementById("subtotal").textContent = "Rs 0";
    document.getElementById("discount").textContent = "Rs 0";
    document.getElementById("cartTotal").textContent = "Rs 0";

    return;
  }

  let subtotal = 0;

  cartItems.innerHTML = cart.map(item => {

    subtotal += item.price * item.qty;

    return `
      <div class="border rounded-lg p-3 flex justify-between items-start">
        <div>
          <h4 class="font-semibold">${item.name}</h4>
          <p class="text-sm text-gray-500">Qty: ${item.qty}</p>
          <p class="font-bold text-blue-600">Rs ${item.price}</p>
        </div>

        <button
          onclick="removeItem(${item.id})"
          class="bg-red-500 hover:bg-red-600 text-white px-3 py-1 rounded">
          Remove
        </button>
      </div>
    `;

  }).join("");

  const discount = subtotal * discountPercent / 100;
  const total = subtotal - discount;

  document.getElementById("subtotal").textContent =
    "Rs " + subtotal.toFixed(2);

  document.getElementById("discount").textContent =
    "- Rs " + discount.toFixed(2);

  document.getElementById("cartTotal").textContent =
    "Rs " + total.toFixed(2);
}

<script>
function openCheckout() {
  if (cart.length === 0) {
    alert("Your cart is empty.");
    return;
  }

  document.getElementById("checkoutModal").classList.remove("hidden");
  document.getElementById("checkoutModal").classList.add("flex");
}

function closeCheckout() {
  document.getElementById("checkoutModal").classList.add("hidden");
  document.getElementById("checkoutModal").classList.remove("flex");
}

function placeOrder() {

  const name = document.getElementById("customerName").value.trim();
  const address = document.getElementById("customerAddress").value.trim();

  if (!name || !address) {
    alert("Please fill all checkout details.");
    return;
  }

  /* Clear Cart */
  cart = [];
  discountPercent = 0;

  document.getElementById("promoCode").value = "";
  document.getElementById("promoMsg").innerHTML = "";

  renderCart();

  /* Reset badge */
  cartCount.textContent = "0";

  /* Close Cart & Checkout */
  closeCheckout();
  closeCart();

  /* Reset Form */
  document.getElementById("customerName").value = "";
  document.getElementById("customerAddress").value = "";

  /* Show Success Popup */
  document.getElementById("successPopup").classList.remove("hidden");
  document.getElementById("successPopup").classList.add("flex");
}

function closeSuccess() {
  document.getElementById("successPopup").classList.add("hidden");
  document.getElementById("successPopup").classList.remove("flex");
}
</script>
